// Tag limit + overflow pill  drawer
selectR('#multi1', {
  mode: 'multiple',
  marker: 'checkbox',
  tagLimit: 2,          // show 2 tags, rest become "+N More Items Selected" pill
  showDrawer: true,     // drawer opens when pill is clicked

});

// Summary pill (collapses to "N Items Selected" once threshold hit)
selectR('#multi2', {
  mode: 'multiple',
  marker: 'circle',
  summaryThreshold: 3,  // collapse to summary pill when >= 3 selected
  summaryLabel: 'Items Selected',
  showDrawer: true,     // drawer lists all selected + individual remove + Select/Clear All
});

8. Server-side Search (AJAX) simulated

Type to search results are fetched from a simulated endpoint (intercepted locally for demo).

No value selected
None selected
// Vanilla JS Implementation
const el = document.getElementById('ajaxSingle');

new SelectR(el, {
  infiniteScroll: true,
  pageSize: 20, 
  ajax: {
    url: '/api/options', 
    method: 'GET',
    data: function(params) {
      return {
        q: params.query, 
        page: params.page,
        pageSize: 20 
      };
    },
    processResults: function(response) {
      return {
        results: response.data, 
        hasMore: response.hasMore
      };
    }
  }
});

13. Initialization & API Examples

1. Initialize:

const el = document.getElementById('my-select');
const mySelect = new SelectR(el, { mode: 'multiple' });

2. Call Methods:

// Use the saved instance...
mySelect.setValue(['1', '2']);
mySelect.open();

// ...or retrieve it from the DOM element later:
const instance = document.getElementById('my-select')._selectR;
instance.clearAll();

1. Initialize:

$('#my-select').selectR({ 
  mode: 'multiple' 
});

2. Call Methods:

// jQuery plugin returns the jQuery object, not the instance.
// To call methods, grab the instance from the raw DOM node:

var instance = $('#my-select')[0]._selectR;

instance.setValue(['1', '2']);
instance.selectAll();

1. Initialize (HTML Only):

<select id="myAutoSelect" data-selectr='{"marker":"tick"}'>...</select>

2. Call Methods:

// The library automatically creates the instance and attaches it to the DOM element.
// You can just grab it and use it anytime!

// Vanilla JS:
document.getElementById('myAutoSelect')._selectR.clearAll();

// jQuery:
$('#myAutoSelect')[0]._selectR.clearAll();

1. Destroying an instance:

// Grabs the instance and completely removes all selectR styling and event listeners, 
// reverting it back to a standard HTML <select>
document.getElementById('mySelect')._selectR.destroy();

2. Reinitializing:

// Once destroyed, you can simply initialize it again normally
const mySelect = new SelectR(document.getElementById('mySelect'), { mode: 'multiple' });

// Or with jQuery:
$('#mySelect').selectR({ mode: 'multiple' });

Initialize dynamically whenever you need to:

// Define your reusable setup function
function initMySelects(selector) {
  // If using jQuery:
  $(selector).selectR({ 
    mode: 'multiple', 
    marker: 'checkbox' 
  });
  
  // Or if using Vanilla JS:
  // document.querySelectorAll(selector).forEach(el => new SelectR(el, { mode: 'multiple' }));
}

// Call it on page load
initMySelects('.my-custom-selects');

// Call it again later if you inject new HTML dynamically
// initMySelects('#my-newly-added-select');

Because selectR syncs with the native HTML element, you have 3 easy ways to get values:

// 1. Standard HTML / jQuery (Recommended)
// Since selectR updates the hidden <select>, you can just read it normally!
var val = document.getElementById('mySelect').value;
var jqueryVal = $('#mySelect').val(); // Returns string (single) or array (multiple)

// 2. Using the selectR API
var apiVal = document.getElementById('mySelect')._selectR.getValue();

// 3. Using the onChange callback during initialization
$('#mySelect').selectR({
  onChange: function(value, selectedItems) {
    console.log("Value changed to:", value);
    // 'value' is a string for single-select, or an array of strings for multiple-select
  }
});

Use the setValue() method to change the selection via JavaScript:

// 1. Grab the instance
var instance = document.getElementById('mySelect')._selectR;
// or jQuery: var instance = $('#mySelect')[0]._selectR;

// 2. Set a Single Value (pass a string)
instance.setValue('apple');

// 3. Set Multiple Values (pass an array of strings)
instance.setValue(['apple', 'banana']);
